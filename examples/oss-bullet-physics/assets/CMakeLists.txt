cmake_minimum_required(VERSION 3.20)
project(SiljangnimGallery LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_CPU_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET2_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_OPENGL3_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET3 OFF CACHE BOOL "" FORCE)
set(BUILD_PYBULLET OFF CACHE BOOL "" FORCE)
set(BUILD_ENET OFF CACHE BOOL "" FORCE)
set(BUILD_CLSOCKET OFF CACHE BOOL "" FORCE)
set(USE_GLUT OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(INSTALL_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${TOOL_SOURCE}/" tool EXCLUDE_FROM_ALL)
add_executable(gallery main.cpp)
target_compile_options(gallery PRIVATE -UNDEBUG)
target_include_directories(gallery PRIVATE "${TOOL_SOURCE}/include" "${TOOL_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/tool/include")
target_link_libraries(gallery PRIVATE BulletDynamics BulletCollision LinearMath)
