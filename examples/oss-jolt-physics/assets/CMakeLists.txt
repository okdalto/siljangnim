cmake_minimum_required(VERSION 3.20)
project(SiljangnimGallery LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TARGET_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(TARGET_HELLO_WORLD OFF CACHE BOOL "" FORCE)
set(TARGET_PERFORMANCE_TEST OFF CACHE BOOL "" FORCE)
set(TARGET_SAMPLES OFF CACHE BOOL "" FORCE)
set(TARGET_VIEWER OFF CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(JPH_USE_MTL OFF CACHE BOOL "" FORCE)
set(JPH_USE_VK OFF CACHE BOOL "" FORCE)
set(JPH_USE_DX12 OFF CACHE BOOL "" FORCE)
set(JPH_USE_CPU_COMPUTE OFF CACHE BOOL "" FORCE)
add_subdirectory("${TOOL_SOURCE}/Build" tool EXCLUDE_FROM_ALL)
add_executable(gallery main.cpp)
target_compile_options(gallery PRIVATE -UNDEBUG)
target_include_directories(gallery PRIVATE "${TOOL_SOURCE}" "${TOOL_SOURCE}/include" "${TOOL_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/tool/include")
target_link_libraries(gallery PRIVATE Jolt)
