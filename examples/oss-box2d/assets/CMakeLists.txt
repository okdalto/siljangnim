cmake_minimum_required(VERSION 3.20)
project(SiljangnimGallery LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BOX2D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX2D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX2D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX2D_DOCS OFF CACHE BOOL "" FORCE)
add_subdirectory("${TOOL_SOURCE}/" tool EXCLUDE_FROM_ALL)
add_executable(gallery main.cpp)
target_compile_options(gallery PRIVATE -UNDEBUG)
target_include_directories(gallery PRIVATE "${TOOL_SOURCE}" "${TOOL_SOURCE}/include" "${TOOL_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/tool/include")
target_link_libraries(gallery PRIVATE box2d)
